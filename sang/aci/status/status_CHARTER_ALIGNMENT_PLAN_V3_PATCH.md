## v3 Patch: Phase 8.1 — Revised `EvaluationReport` schema

Replace the `EvaluationReport` class body in `src/jaci/schemas/evaluation_report.py`
with the following. The key change is splitting `scores` (Python-computed) from the
LLM-generated fields (`sar_quality_score`, `sar_quality_assessment`,
`improvement_signals`), making the source of each field explicit in the schema.

```python
class EvaluationReport(BaseModel):
    """
    Formally typed II EvaluationReport for a JACI-AML case.

    Field sources:
      PYTHON-COMPUTED (deterministic, no LLM, fast):
        scores — decision_accuracy, evidence_efficiency, policy_compliance, loop_efficiency
        findings — auto-generated for any score below FINDING_THRESHOLD (0.7)
        human_override_detected — from Outcome.human_override

      LLM-GENERATED (qualitative, one LLM call per case):
        sar_quality_score — 0.0–1.0, Escalate cases only (None for Close)
        sar_quality_assessment — prose explanation of SAR quality score
        improvement_signals — bracket-tagged actionable signals for Curator
    """

    report_id: str = Field(
        default_factory=lambda: f"evalreport_{uuid4()}",
        description="Namespaced UUID. Format: evalreport_{uuid_v4}.",
    )
    evaluation_type: str = Field(default="periodic_review")

    # Subject references
    case_id: str = Field(...)
    decision_id: str = Field(...)
    trace_id: str = Field(...)
    outcome_id: str = Field(...)

    # Evaluation period
    period_start: datetime = Field(...)
    period_end: datetime = Field(...)

    # ── PYTHON-COMPUTED (deterministic) ──────────────────────────────────
    scores: dict[str, float] = Field(
        default_factory=dict,
        description=(
            "Deterministic metrics computed by compute_deterministic_metrics(). "
            "Keys: decision_accuracy, evidence_efficiency, policy_compliance, "
            "loop_efficiency. Values: 0.0–1.0. "
            "These are NEVER set by the LLM."
        ),
    )
    findings: list[str] = Field(
        default_factory=list,
        description=(
            "Auto-generated finding for each score in `scores` below 0.7. "
            "Format: '[dimension]: <Python-derived observation>'. "
            "Generated by compute_deterministic_metrics(), not by LLM."
        ),
    )
    human_override_detected: bool = Field(
        default=False,
        description=(
            "True if Outcome.human_override=True. "
            "Set by Python from Outcome, not by LLM."
        ),
    )

    # ── LLM-GENERATED (qualitative) ──────────────────────────────────────
    sar_quality_score: float | None = Field(
        default=None,
        ge=0.0,
        le=1.0,
        description=(
            "SAR narrative quality score 0.0–1.0. "
            "Set by LLM. None for Close cases (no SAR draft). "
            "Never set by Python."
        ),
    )
    sar_quality_assessment: str | None = Field(
        default=None,
        description=(
            "LLM prose explanation of the SAR quality score. "
            "None for Close cases. "
            "Never set by Python."
        ),
    )
    improvement_signals: list[str] = Field(
        default_factory=list,
        description=(
            "Bracket-tagged actionable improvement signals generated by LLM. "
            "Format: '[tag] description' where tag is one of VALID_SIGNAL_TAGS. "
            "Routed by Curator to typed Knowledge Hub queue buckets. "
            "Never set by Python."
        ),
    )

    # Governance
    pack_id: str = Field(default=PACK_ID)
    created_at: datetime = Field(default_factory=datetime.utcnow)
```

Also export `FINDING_THRESHOLD = 0.7` as a module-level constant so
`compute_deterministic_metrics()` and tests reference the same value.

Every improvement signal MUST start with one of these six bracket tags.

## v3 Patch: Phase 9.2 — Add `VALID_SIGNAL_TAGS` to `curator_stub.py`

Replace the `_TAG_PATTERN` and `_extract_tag` section in `curator_stub.py` with:

```python
# Closed set of valid signal tags — must match the tag set in prompts/evaluator.md.
# Update both files if the tag set changes.
VALID_SIGNAL_TAGS = frozenset({
    "evidence_checklist",
    "typology_threshold",
    "policy_clause",
    "loop_guard",
    "sar_template",
    "other",
})

# Bracket tag pattern: [tag_name] at start of signal string
_TAG_PATTERN = re.compile(r"^\[([a-z_]+)\]")


def _extract_tag(signal: str) -> str:
    """
    Extract bracket tag from signal string.

    Returns the tag if it's in VALID_SIGNAL_TAGS, otherwise 'other'.
    Unrecognised tags are rerouted to 'other' — signals are never dropped.

    Examples:
      '[evidence_checklist] Depth 3 needed for shell companies' → 'evidence_checklist'
      '[unknown_tag] Some signal' → 'other'
      'Signal without tag' → 'other'
    """
    match = _TAG_PATTERN.match(signal.strip())
    if match:
        tag = match.group(1)
        return tag if tag in VALID_SIGNAL_TAGS else "other"
    return "other"
```

"""
Unit tests for compute_deterministic_metrics().

Tests run without LLM — no API key required.
compute_deterministic_metrics() is pure Python; all inputs are constructed directly.
"""

import pytest
from datetime import datetime
from uuid import uuid4

# These tests use lightweight mock objects rather than full CaseFile/Outcome
# instances, since we only care about the specific fields the function reads.


class MockEvidence:
    def __init__(self, evidence_id, verifier_status="attested", cited=True):
        self.evidence_id = evidence_id
        self.verifier_status = verifier_status
        self._cited = cited


class MockCaseContext:
    def __init__(self, evidence, iteration_count):
        self.evidence = evidence
        self.iteration_count = iteration_count

    @property
    def attested_evidence(self):
        return [e for e in self.evidence if e.verifier_status == "attested"]


class MockDisposition:
    def __init__(self, recommendation, supporting_evidence_ids, policy_refs):
        self.recommendation = recommendation
        self.supporting_evidence_ids = supporting_evidence_ids
        self.policy_refs = policy_refs


class MockCaseFile:
    def __init__(self, disposition, case_context):
        self.disposition = disposition
        self.case_context = case_context


class MockOutcome:

        self.outcome_id = f"outcome_{uuid4()}"


# ─── decision_accuracy ───────────────────────────────────────────────────────

class TestDecisionAccuracy:

    def test_perfect_match_no_override(self):
        """No override → 1.0 regardless of evidence count."""
        from jaci.modes.evaluator import compute_deterministic_metrics
        evidence = [MockEvidence(f"ev_{i}") for i in range(3)]
        case_file = MockCaseFile(
            disposition=MockDisposition("escalate", [e.evidence_id for e in evidence], ["BSA-CTR-001"]),
            case_context=MockCaseContext(evidence, iteration_count=3),
        )
        outcome = MockOutcome(human_override=False)

**Note on `return_findings` parameter:** The test suite uses `return_findings=True`
to get findings separately from scores. Implement `compute_deterministic_metrics()`
to support this optional parameter:

```python
def compute_deterministic_metrics(
    case_file: CaseFile,
    outcome: Outcome,
    return_findings: bool = False,
) -> dict[str, float] | tuple[dict[str, float], list[str]]:
    """
    ...existing docstring...

    Args:
        case_file: CaseFile with disposition and case_context.
        outcome: Outcome from L3/FIU review.
        return_findings: If True, return (scores, findings) tuple.
                         Default False returns scores dict only.
    """
    scores = {}
    findings = []
    # ... existing computation ...

    # Auto-generate findings for any score below threshold
    for dimension, score in scores.items():
        if score < FINDING_THRESHOLD:
            findings.append(
                f"[{dimension}]: score {score:.2f} below threshold {FINDING_THRESHOLD}"
            )

    return (scores, findings) if return_findings else scores
```
